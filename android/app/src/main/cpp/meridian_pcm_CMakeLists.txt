# Built from android/scripts/build-media3-ffmpeg.sh (not the aubio Gradle CMake).
# Expects:
#   MERIDIAN_PCM_SRC  — path to meridian_pcm_jni.c
#   MERIDIAN_FFMPEG_LIBDIR — android-libs/<abi> with libav*.a
#   MERIDIAN_FFMPEG_INCLUDE — ffmpeg source tree (for headers)
cmake_minimum_required(VERSION 3.22.1)
project(meridian_pcm C)

add_library(meridian_pcm SHARED ${MERIDIAN_PCM_SRC})

target_include_directories(meridian_pcm PRIVATE
    ${MERIDIAN_FFMPEG_INCLUDE}
    ${MERIDIAN_FFMPEG_LIBDIR}/../include
)

# Headers live in the ffmpeg source tree; installed libs in android-libs/<abi>.
target_include_directories(meridian_pcm PRIVATE
    ${MERIDIAN_FFMPEG_INCLUDE}
)

find_library(android_log_lib log)
find_library(android_z_lib z)
find_library(android_m_lib m)

# Order matters for static archives.
target_link_libraries(meridian_pcm
    ${MERIDIAN_FFMPEG_LIBDIR}/libavformat.a
    ${MERIDIAN_FFMPEG_LIBDIR}/libavcodec.a
    ${MERIDIAN_FFMPEG_LIBDIR}/libswresample.a
    ${MERIDIAN_FFMPEG_LIBDIR}/libavutil.a
    ${android_log_lib}
    ${android_z_lib}
    ${android_m_lib}
)

# Same NDK 23+ arm64 fix Media3 uses for libffmpegJNI.so (NEON asm in avutil).
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_link_options(meridian_pcm PRIVATE "-Wl,-Bsymbolic")
endif()

target_compile_options(meridian_pcm PRIVATE -O2 -fPIC -std=c11)
